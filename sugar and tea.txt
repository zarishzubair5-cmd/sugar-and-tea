<script>
function sendOrder() {
  const name = document.getElementById("name").value;
  const milkChoice = document.querySelector('input[name="milk"]:checked');

  if (!name || !milkChoice) {
    alert("Please complete your order 💗");
    return;
  }

  fetch("PASTE_SCRIPT_URL_HERE", {
    method: "POST",
    body: JSON.stringify({
      name: name,
      drink: "Matcha Latte",
      milk: milkChoice.value
    })
  });

  alert("Order sent to Sugar & Tea 🍵");
}
